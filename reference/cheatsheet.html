⚡ Quick Reference Cheat Sheet

Pin-printable reference for Spring Boot JMS + Profiles syntax. All code is Spring Boot 3 + Artemis.

1. Maven Dependencies

<!-- ActiveMQ Artemis (Spring Boot 3) -->
<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-artemis</artifactId>
</dependency>

<!-- Embedded broker (dev/test only) -->
<dependency>
    <groupId>org.apache.activemq</groupId>
    <artifactId>artemis-server</artifactId>
    <scope>test</scope>
</dependency>

2. Application Properties

application.yml (shared)

spring:
  artemis:
    mode: embedded          # or 'native' for external broker
    broker-url: tcp://localhost:61616
    user: admin
    password: admin
  jms:
    pub-sub-domain: false   # false=queues (default), true=topics

application-dev.yml

spring:
  artemis:
    mode: embedded
logging:
  level:
    org.apache.activemq: DEBUG

application-prod.yml

spring:
  artemis:
    mode: native
    broker-url: tcp://broker.prod.internal:61616
    user: ${ARTEMIS_USER}
    password: ${ARTEMIS_PASSWORD}
  jms:
    cache:
      enabled: true
      session-cache-size: 10

3. Sending Messages

Simple text message

@Service
public class NotificationSender {

    private final JmsTemplate jmsTemplate;

    public NotificationSender(JmsTemplate jmsTemplate) {
        this.jmsTemplate = jmsTemplate;
    }

    public void send(String message) {
        jmsTemplate.convertAndSend("notification.queue", message);
    }
}

POJO as JSON

// Requires MessageConverter bean (see section 5)
jmsTemplate.convertAndSend("order.queue", new Order("ORD-001", 299.99));

With custom headers

jmsTemplate.convertAndSend("order.queue", order, msg -> {
    msg.setStringProperty("region", "APAC");
    msg.setIntProperty("priority", 5);
    return msg;
});

4. Receiving Messages

Basic listener

@Component
public class NotificationListener {

    @JmsListener(destination = "notification.queue")
    public void onMessage(String message) {
        System.out.println("Received: " + message);
    }
}

POJO listener (auto-deserialized)

@JmsListener(destination = "order.queue")
public void onOrder(Order order) {
    System.out.println("Processing: " + order.getId());
}

With message selector

@JmsListener(
    destination = "order.queue",
    selector = "region = 'APAC' AND priority > 3"
)
public void onHighPriorityApac(Order order) { ... }

Topic listener

@JmsListener(
    destination = "price.updates",
    containerFactory = "topicFactory"
)
public void onPriceUpdate(PriceEvent event) { ... }

5. Configuration Beans

JSON MessageConverter

@Bean
public MessageConverter jacksonJmsMessageConverter() {
    MappingJackson2MessageConverter converter = new MappingJackson2MessageConverter();
    converter.setTargetType(MessageType.TEXT);
    converter.setTypeIdPropertyName("_type");
    return converter;
}

Topic ListenerContainerFactory

@Bean
public JmsListenerContainerFactory<?> topicFactory(
        ConnectionFactory cf,
        DefaultJmsListenerContainerFactoryConfigurer configurer) {
    DefaultJmsListenerContainerFactory factory = new DefaultJmsListenerContainerFactory();
    configurer.configure(factory, cf);
    factory.setPubSubDomain(true);
    return factory;
}

CachingConnectionFactory (production)

@Bean
@Profile("prod")
public ConnectionFactory cachingConnectionFactory(
        ConnectionFactory artemisConnectionFactory) {
    CachingConnectionFactory factory = new CachingConnectionFactory(artemisConnectionFactory);
    factory.setSessionCacheSize(10);
    return factory;
}

6. Spring Profiles

Activate via CLI

java -jar app.jar --spring.profiles.active=prod

Activate via environment variable

export SPRING_PROFILES_ACTIVE=dev
./mvnw spring-boot:run

Activate in Maven

./mvnw spring-boot:run -Dspring-boot.run.profiles=dev

Profile-specific bean

@Service
@Profile("dev")
public class MockPaymentService implements PaymentService {
    // Fake payment processing for development
}

@Service
@Profile("prod")
public class StripePaymentService implements PaymentService {
    // Real Stripe integration for production
}

Profile groups

spring:
  profiles:
    group:
      production: "prod,metrics,ssl"
      development: "dev,debug,embedded-broker"

Negated profile

@Profile("!prod")  // Active in all profiles EXCEPT prod

7. Docker Commands

Run Artemis

# Run ActiveMQ Artemis in Docker
docker run -d --name artemis \
  -p 61616:61616 \
  -p 8161:8161 \
  -e ARTEMIS_USER=admin \
  -e ARTEMIS_PASSWORD=admin \
  apache/activemq-artemis:latest

# Web console: http://localhost:8161
# Broker URL:  tcp://localhost:61616
